# Different compiler warning levels can be set in the Arduino IDE/CLI
# This is currently set here, but not used, as the CODAL build system needs to be hooked
# to take some flag defined here into the CMakeLists.txt file and apply flags appropriately
compiler.warning_flags=-w

compiler.warning_flags.default=

# Arduino macros that need to be defined globally, e.g. F_CPU is used by some libraries for timing calculations
compiler.arduino.flags=-DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch}

# Compiler general flags for compiling C/C++ code for this platform
# Only really used for the preprocessor, as everything else is compiled with the CODAL build system
compiler.general.flags=-mcpu={build.mcu} -mthumb -c {compiler.warning_flags} -std=gnu++11 -ffunction-sections -fdata-sections -nostdlib -fno-rtti -fno-exceptions

# Capture all the CODAL include directories, so that it can be included by the preprocessor
compiler.codal.flags=-DNRF52833_XXAA
compiler.codal.includes_microbit_v2="-I{build.system.path}/codal/libraries/codal-microbit-v2/inc" "-I{build.system.path}/codal/libraries/codal-microbit-v2/inc/compat/" "-I{build.system.path}/codal/libraries/codal-microbit-v2/model" "-I{build.system.path}/codal/libraries/codal-microbit-v2/inc/bluetooth/"
compiler.codal.includes_core="-I{build.system.path}/codal/libraries/codal-core/inc/core" "-I{build.system.path}/codal/libraries/codal-core/inc/driver-models" "-I{build.system.path}/codal/libraries/codal-core/inc/drivers" "-I{build.system.path}/codal/libraries/codal-core/inc/streams" "-I{build.system.path}/codal/libraries/codal-core/inc/types" "-I{build.path}/codal/libraries/codal-core/gen"
compiler.codal.includes_nrf52="-I{build.system.path}/codal/libraries/codal-nrf52/inc/" "-I{build.system.path}/codal/libraries/codal-nrf52/inc/cmsis"
compiler.codal.includes_nrfx="-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/mdk/" "-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/" "-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/templates/" "-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/templates/nRF52833/" "-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/drivers/include" "-I{build.system.path}/codal/libraries/codal-nrf52/nrfx/drivers/src"
compiler.codal.includes_codal="-I{build.system.path}/codal/utils/cmake/toolchains/ARM_GCC/" "-I{build.system.path}/codal/libraries/" -include "{build.path}/codal/codal_extra_definitions_new.h" -include "{build.path}/codal/codal_extra_definitions.h"
compiler.codal.includes={compiler.codal.includes_codal} {compiler.codal.includes_core} {compiler.codal.includes_nrfx} {compiler.codal.includes_nrf52} {compiler.codal.includes_microbit_v2}

# The preoprocesor runs first to generate function prototypes, convert the ino to a cpp file, etc
# https://arduino.github.io/arduino-cli/0.27/platform-specification/?utm_source=pocket_mylist#recipe-to-run-the-preprocessor
# The main thing needed for the preprocessor flags is -E, so that it generates the function prototypes correctly
preproc.macros.flags={compiler.general.flags} {compiler.codal.flags} {compiler.arduino.flags} "-I{build.core.path}" "-I{build.core.path}/api" "-I{build.core.path}/api/deprecated" "-I{build.core.path}/api/deprecated-avr-comp/avr" -E -w -x c++ -CC
# If recipe.preproc.macros is not defined, it is automatically generated from recipe.cpp.o.pattern.
recipe.preproc.macros= "{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {preproc.macros.flags} {compiler.cpp.extra_flags} {build.extra_flags} {compiler.codal.includes} {includes} "{source_file}" -o "{preprocessed_file_path}"

# We don't compile with the Arduino build system, we'll use CMake in a hook later on.
# Instead, the compile recipes record each source file the Arduino CLI detected (sketch
# and Arduino libraries, e.g. installed via the Library Manager) plus its include paths
# into a manifest folder, which the CODAL CMake build then reads (arduino/manifest-read.cmake)
# to compile and link those files together with everything else.
build.arduino_manifest={build.path}/arduino-manifest
recipe.c.o.pattern={tools.cmake.cmd} "-DMANIFEST_DIR={build.arduino_manifest}" -P "{runtime.platform.path}/system/arduino-manifest/manifest-write.cmake" -- "{source_file}" {includes}
recipe.cpp.o.pattern={tools.cmake.cmd} "-DMANIFEST_DIR={build.arduino_manifest}" -P "{runtime.platform.path}/system/arduino-manifest/manifest-write.cmake" -- "{source_file}" {includes}
recipe.S.o.pattern={tools.cmake.cmd} "-DMANIFEST_DIR={build.arduino_manifest}" -P "{runtime.platform.path}/system/arduino-manifest/manifest-write.cmake" -- "{source_file}" {includes}
# Deliberate no-op, core objects are compiled by the CODAL CMake build via manifest.
# If this step was ever implemented here, subsequent builds would skip compiling unchanged
# core files, leaving them out of the manifest and break the CODAL CMake build.
recipe.ar.pattern={tools.cmake.cmd} -E echo "Core archiving skipped (CODAL CMake compiles the core)"

# The linking is done by CMake/Ninja in the postlink hooks; this recipe only records the
# link flags of "precompiled" Arduino libraries (empty unless such a library is used)
recipe.c.combine.pattern={tools.cmake.cmd} "-DMANIFEST_DIR={build.arduino_manifest}" -DENTRY_TYPE=ldflags -P "{runtime.platform.path}/system/arduino-manifest/manifest-write.cmake" -- {compiler.libraries.ldflags}

# CODAL's toolchain.cmake locates arm-none-eabi-gcc via find_program(),
# with this flag we add an extra path with higher priority
codal.toolchain.flags="-DCMAKE_PROGRAM_PATH={compiler.path}"

# Create the CODAL build files before the Arduino build system starts preprocessing
# This is to ensure codal_extra_definitions.h and codal_extra_definitions_new.h are created
# Runs before the Arduino manifest is cleared (prebuild.2), as it prunes ninja outputs no
# longer in the build, so it must still see the previous build's manifest sources to keep them
recipe.hooks.prebuild.1.pattern={tools.cmake.cmd} -S "{build.system.path}/codal/" -B "{build.path}/codal/" "-DCMAKE_MAKE_PROGRAM={tools.ninja.program}" {codal.toolchain.flags} -DCMAKE_BUILD_TYPE=RelWithDebInfo -G "Ninja"
# Then clear the manifest of any previous build before the compile recipes repopulate it
recipe.hooks.prebuild.2.pattern={tools.cmake.cmd} -E rm -rf "{build.arduino_manifest}"
# The first time the CODAL build system is run the Arduino manifest is not yet populated
# by the compile recipes, so run it again at linking stage with the manifest and Arduino defines
recipe.hooks.linking.postlink.1.pattern={tools.cmake.cmd} -S "{build.system.path}/codal/" -B "{build.path}/codal/" "-DCMAKE_MAKE_PROGRAM={tools.ninja.program}" {codal.toolchain.flags} "-DF_CPU={build.f_cpu}" "-DARDUINO={runtime.ide.version}" "-DARDUINO_BOARD=ARDUINO_{build.board}" "-DARDUINO_ARCH=ARDUINO_ARCH_{build.arch}" "-DARDUINO_LIBRARIES={runtime.platform.path}/libraries" "-DARDUINO_VARIANT={build.variant.path}" "-DARDUINO_LIBRARIES_MANIFEST={build.arduino_manifest}" -DCMAKE_BUILD_TYPE=RelWithDebInfo -G "Ninja"
# And finally do the real compilation as a second post link hook
recipe.hooks.linking.postlink.2.pattern={tools.ninja.cmd} -C "{build.path}/codal/"

# Now that CMake has built everything, create the hex and bin files from the ELF file
recipe.objcopy.bin.pattern={compiler.path}{compiler.objcopy.cmd} -O binary "{build.path}/codal/MICROBIT" "{build.path}/{build.project_name}.bin"
recipe.objcopy.hex.pattern={compiler.path}{compiler.objcopy.cmd} -O ihex  "{build.path}/codal/MICROBIT" "{build.path}/{build.project_name}.hex"
